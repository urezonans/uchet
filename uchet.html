function openTimesheetMonthSummary(empId, year, month) { const emp = employees.find(e => e.id === empId); if (!emp) return; const daysInMonth = new Date(year, month + 1, 0).getDate(); document.getElementById('tsMonthSummaryEmployee').textContent = emp.name; document.getElementById('tsMonthSummaryPeriod').textContent = MONTH_NAMES[month] + ' ' + year; // Собираем все записи за месяц const projectHours = {}; let totalHours = 0; for (let day = 1; day <= daysInMonth; day++) { const dateStr = year + '-' + String(month + 1).padStart(2, '0') + '-' + String(day).padStart(2, '0'); const key = getTimesheetKey(empId, dateStr); const entry = timesheets[key]; if (entry && entry.entries) { entry.entries.forEach(e => { if (!projectHours[e.projectId]) { projectHours[e.projectId] = { projectId: e.projectId, projectTitle: e.projectTitle, clientName: e.clientName, hours: 0, days: 0 }; } projectHours[e.projectId].hours += e.hours; projectHours[e.projectId].days += 1; totalHours += e.hours; }); } } const listEl = document.getElementById('tsMonthSummaryList'); const projects = Object.values(projectHours); if (!projects.length) { listEl.innerHTML = '
Нет записей за этот месяц
'; } else { // Сортируем по часам (убывание) projects.sort((a, b) => b.hours - a.hours); listEl.innerHTML = projects.map(p => { const percentage = totalHours > 0 ? ((p.hours / totalHours) * 100).toFixed(1) : 0; return `
${escapeHtml(p.projectTitle)}
${escapeHtml(p.clientName)} • ${p.days} дн.
${p.hours.toFixed(1)} ч
${percentage}%
`; }).join(''); } document.getElementById('tsMonthSummaryTotal').textContent = totalHours.toFixed(1); document.getElementById('timesheetMonthSummaryModal').style.display = 'block'; } function closeTimesheetMonthSummary() { document.getElementById('timesheetMonthSummaryModal').style.display = 'none'; }
